    <script>
        let totalMinutes = 600;

        function updateClock() {
            const hours = Math.floor(totalMinutes / 60) % 12;
            const mins = totalMinutes % 60;
            const displayHours = Math.floor(totalMinutes / 60) % 24;

            const hDeg = (hours * 30) + (mins * 0.5);
            const mDeg = mins * 6;
            
            document.getElementById('hour-hand').style.transform = `rotate(${hDeg}deg)`;
            document.getElementById('minute-hand').style.transform = `rotate(${mDeg}deg)`;
            
            const hStr = displayHours === 0 ? "12" : (displayHours > 12 ? displayHours - 12 : displayHours);
            const mStr = mins.toString().padStart(2, '0');
            document.getElementById('digital-display').innerText = `${hStr}:${mStr}`;
        }

        function addMinutes(m) {
            totalMinutes += m;
            if (totalMinutes < 0) totalMinutes += 1440;
            updateClock();
        }

        function createFace() {
            const face = document.getElementById('clock-face');
            const radius = face.offsetWidth / 2 - 25; // 縁からの距離を調整
            const centerX = face.offsetWidth / 2;
            const centerY = face.offsetHeight / 2;

            for (let i = 1; i <= 12; i++) {
                const num = document.createElement('div');
                num.className = 'number';
                num.innerText = i;
                const angle = (i * 30) * (Math.PI / 180);
                const x = centerX + radius * Math.sin(angle) - 15;
                const y = centerY - radius * Math.cos(angle) - 15;
                num.style.left = x + 'px';
                num.style.top = y + 'px';
                face.appendChild(num);
            }
        }

        window.onload = () => {
            createFace();
            updateClock();
        };

        // 画面サイズが変わった時に数字の位置を再計算（スマホの回転対策）
        window.onresize = () => {
            const face = document.getElementById('clock-face');
            face.innerHTML = '<div class="center-dot"></div><div id="hour-hand" class="hand hour-hand"></div><div id="minute-hand" class="hand minute-hand"></div>';
            createFace();
            updateClock();
        };
    </script>
